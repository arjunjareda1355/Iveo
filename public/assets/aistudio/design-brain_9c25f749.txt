# Iveo Design Brain

Curated, rule-based theme generator for Appearance → "Surprise me", the idea rows, and the refresh icons.
Zero dependencies. Plain TypeScript. Copy the folder to `/lib/design-brain/`.

## Files
| File | Purpose |
|---|---|
| util.ts | colour maths, WCAG contrast, seeded RNG |
| data-palettes.ts / data-vibes.ts / data-fonts.ts / data-styles.ts | **All the design knowledge. Edit these to add palettes, vibes, font trios, button recipes, layouts.** |
| build.ts | turns a palette into background / buttons / layout values |
| rules.ts | HARD rules (never broken) + SOFT score 0-100 |
| generator.ts | generateTheme, suggestSection, regenerate, repair, locks |
| api.ts | handleSurprise / handleSuggestions / handleRegenerate (framework-agnostic) |
| adapter.ts | **the only file to edit to match your DB schema** |
| selftest.ts, preview.ts | 1000-theme test, 24-theme visual grid |

## Wire-up (3 steps)
1. Edit `adapter.ts` so `toIveoAppearance()` returns your saved-appearance shape (and the reverse when loading `current`).
2. Expose routes, e.g. Next.js:
   ```ts
   // app/api/appearance/surprise/route.ts
   import { handleSurprise } from '@/lib/design-brain/api';
   export async function GET(req: Request) {
     const q = Object.fromEntries(new URL(req.url).searchParams);
     const ctx = await buildContextFromUser(); // { plan, photoColors, bio, linkTypes, current, locked }
     const r = handleSurprise(q, ctx); return Response.json(r.body, { status: r.status });
   }
   ```
   Same pattern for `/suggestions?section=` (handleSuggestions) and POST `/regenerate` (handleRegenerate).
3. Frontend: "Surprise me" → GET surprise, apply `theme` through your existing setters. Idea cards render
   `name`, `tagline`, `badge`+`ratio`, `dots`; clicking one merges `suggestion.patch` into the current theme.
   Keep the last 20 `signatures` client-side and send them as `recent` on refresh.

## Context you pass in (all optional)
`plan` free|pro · `mood` · `useCase` · `photoColors` (hex[] from "From your photo") · `bio` · `linkTypes` (e.g. ["spotify","github"]) ·
`current` (user's theme) · `locked` ({vibe, background, typography, buttons, layout}).

## Guarantees (verified by selftest.ts)
0 contrast failures, 0 forbidden pairings, no display fonts for bio/buttons, ≤3 fonts, neon only on dark, neumorphic only on calm light,
glass only on gradient-like, busy backgrounds force solid/glass buttons, free users never get pro items,
locked sections come back unchanged, same seed = same theme (store `meta.seed` to share/undo).

## Adding combinations
New palette → one object in data-palettes.ts, link it from a vibe's `palettes`. New font pairing → one row in TRIOS.
Then run selftest; it re-checks palette contrast and every rule across 1000 generated themes.

// util.ts: colour maths + seeded RNG (no dependencies)

export type Hex = string;

export function hexToRgb(h: Hex): [number, number, number] {
  const s = h.replace('#', '');
  const n = parseInt(s.length === 3 ? s.split('').map(c => c + c).join('') : s, 16);
  return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
}
export function rgbToHex(r: number, g: number, b: number): Hex {
  const c = (v: number) => Math.max(0, Math.min(255, Math.round(v))).toString(16).padStart(2, '0');
  return ('#' + c(r) + c(g) + c(b)).toUpperCase();
}
export function luminance(h: Hex): number {
  const f = (v: number) => { v /= 255; return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); };
  const [r, g, b] = hexToRgb(h);
  return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b);
}
export function contrast(a: Hex, b: Hex): number {
  const la = luminance(a), lb = luminance(b);
  return (Math.max(la, lb) + 0.05) / (Math.min(la, lb) + 0.05);
}
export function isDark(h: Hex): boolean { return luminance(h) < 0.25; }
export function badge(ratio: number): 'AAA' | 'AA' | 'AA Large' | 'Fail' {
  return ratio >= 7 ? 'AAA' : ratio >= 4.5 ? 'AA' : ratio >= 3 ? 'AA Large' : 'Fail';
}
/** Best readable text colour for a background: tries preferred first, then near-black/white. */
export function readableOn(bg: Hex, preferred?: Hex, min = 4.5): Hex {
  if (preferred && contrast(bg, preferred) >= min) return preferred;
  const dark = '#0B0A0A', light = '#FFFFFF';
  return contrast(bg, dark) >= contrast(bg, light) ? dark : light;
}

export function rgbToHsl(h: Hex): [number, number, number] {
  let [r, g, b] = hexToRgb(h).map(v => v / 255);
  const max = Math.max(r, g, b), min = Math.min(r, g, b);
  let hh = 0, s = 0; const l = (max + min) / 2;
  if (max !== min) {
    const d = max - min;
    s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
    hh = max === r ? (g - b) / d + (g < b ? 6 : 0) : max === g ? (b - r) / d + 2 : (r - g) / d + 4;
    hh *= 60;
  }
  return [hh, s * 100, l * 100];
}
export function hslToHex(h: number, s: number, l: number): Hex {
  h = ((h % 360) + 360) % 360; s = Math.max(0, Math.min(100, s)) / 100; l = Math.max(0, Math.min(100, l)) / 100;
  const k = (n: number) => (n + h / 30) % 12;
  const a = s * Math.min(l, 1 - l);
  const f = (n: number) => l - a * Math.max(-1, Math.min(k(n) - 3, Math.min(9 - k(n), 1)));
  return rgbToHex(f(0) * 255, f(8) * 255, f(4) * 255);
}
/** Safe small hue/lightness nudge: creates variety without breaking the palette. */
export function shift(h: Hex, dHue: number, dSat = 0, dLight = 0): Hex {
  const [hh, s, l] = rgbToHsl(h);
  return hslToHex(hh + dHue, s + dSat, l + dLight);
}
export function mix(a: Hex, b: Hex, t: number): Hex {
  const A = hexToRgb(a), B = hexToRgb(b);
  return rgbToHex(A[0] + (B[0] - A[0]) * t, A[1] + (B[1] - A[1]) * t, A[2] + (B[2] - A[2]) * t);
}

// ---------- seeded RNG (mulberry32) ----------
export function hashSeed(s: string | number): number {
  const str = String(s); let h = 1779033703 ^ str.length;
  for (let i = 0; i < str.length; i++) { h = Math.imul(h ^ str.charCodeAt(i), 3432918353); h = (h << 13) | (h >>> 19); }
  return (h ^ (h >>> 16)) >>> 0;
}
export class RNG {
  private a: number;
  constructor(seed: string | number) { this.a = hashSeed(seed) || 1; }
  next(): number {
    let t = (this.a += 0x6d2b79f5);
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  }
  int(min: number, max: number) { return Math.floor(this.next() * (max - min + 1)) + min; }
  range(min: number, max: number) { return min + this.next() * (max - min); }
  pick<T>(arr: readonly T[]): T { return arr[Math.floor(this.next() * arr.length)]; }
  /** weighted pick: weights[i] >= 0 */
  weighted<T>(arr: readonly T[], weights: number[]): T {
    const total = weights.reduce((a, b) => a + Math.max(0, b), 0);
    if (total <= 0) return this.pick(arr);
    let r = this.next() * total;
    for (let i = 0; i < arr.length; i++) { r -= Math.max(0, weights[i]); if (r <= 0) return arr[i]; }
    return arr[arr.length - 1];
  }
  chance(p: number) { return this.next() < p; }
}

// build.ts: turn a palette + RNG into concrete background / button / layout values
import { RNG, mix, shift, contrast, readableOn, isDark, type Hex } from './util';
import type { Palette, Mood, UseCase } from './data-palettes';
import { BG_RECIPES, SURFACES, MOTIONS, HEADERS, AVATAR_FRAMES, CARD_LOOKS, AVATAR_SIZES,
  type BgRecipe, type SurfaceRecipe, type Opt } from './data-styles';
import type { Theme } from './types';

const ov = (a: readonly string[], b: readonly string[]) => a.filter(x => b.includes(x)).length;
export const weightBy = <T extends { moods: Mood[]; useCases?: UseCase[]; weight?: number }>(
  items: readonly T[], moods: Mood[], useCase?: UseCase) =>
  items.map(i => (i.weight ?? 1) * (0.15 + ov(i.moods, moods)) * (useCase && i.useCases?.includes(useCase) ? 1.8 : 1));

/** Pull a stop toward the base colour until text stays readable on it. */
function fixStop(c: Hex, base: Hex, text: Hex, min = 4.8): Hex {
  let out = c;
  for (let i = 0; i < 6 && contrast(out, text) < min; i++) out = mix(out, base, 0.5);
  return contrast(out, text) < min ? base : out;
}

export function buildBackground(p: Palette, moods: Mood[], rng: RNG, plan: 'free' | 'pro'): Theme['background'] {
  const pool = BG_RECIPES.filter(b => plan === 'pro' || !b.pro);
  const r: BgRecipe = rng.weighted(pool, weightBy(pool, moods));
  const bg = p.bg, bg2 = p.bg2, tx = p.text;
  const angle = rng.int(120, 215), t = rng.range(0.15, 0.4);
  const st = (c: Hex, pos: number) => ({ color: fixStop(c, bg, tx), pos });
  const base = { kind: r.kind, name: r.label, color: bg } as Theme['background'];
  switch (r.kind) {
    case 'solid': return base;
    case 'tonal': return { ...base, color: mix(bg, bg2, t) };
    case 'soft-gradient': return { ...base, gradient: { type: 'linear', angle, stops: [st(bg, 0), st(bg2, 100)] } };
    case 'gradient': return { ...base, gradient: { type: rng.pick(['linear', 'linear', 'radial'] as const), angle: rng.int(0, 360),
      stops: [st(bg, 0), st(mix(bg, p.accent, rng.range(0.12, 0.25)), rng.int(40, 60)), st(bg2, 100)] } };
    case 'mesh': return { ...base, gradient: { type: 'radial', angle: rng.int(0, 360), stops: [
      st(mix(bg, p.accent, rng.range(0.1, 0.22)), 0), st(bg, 35), st(mix(bg, p.accent2, rng.range(0.08, 0.2)), 70), st(bg2, 100)] } };
    case 'aurora': return { ...base, gradient: { type: 'linear', angle: rng.int(150, 210), stops: [
      st(bg, 0), st(mix(bg, p.accent, 0.2), 35), st(mix(bg, p.accent2, 0.16), 65), st(bg2, 100)] } };
    case 'glass': return { ...base, gradient: { type: 'linear', angle, stops: [st(mix(bg, p.surface, 0.5), 0), st(bg2, 100)] } };
    case 'grain': return { ...base, color: mix(bg, bg2, t), grain: +rng.range(0.15, 0.45).toFixed(2) };
    case 'pattern': return { ...base, color: bg, grain: +rng.range(0.05, 0.15).toFixed(2) };
    case 'animated': return { ...base, gradient: { type: 'linear', angle, stops: [st(bg, 0), st(mix(bg, p.accent, 0.18), 50), st(bg2, 100)] } };
  }
}

export function pickSurface(p: Palette, bgk: Theme['background'], moods: Mood[], rng: RNG, plan: 'free' | 'pro'): SurfaceRecipe {
  const br = BG_RECIPES.find(b => b.kind === bgk.kind)!;
  const dark = isDark(bgk.color);
  const ok = SURFACES.filter(s => {
    if (s.pro && plan === 'free') return false;
    const n = s.needs;
    if (n?.darkBg && !dark) return false;
    if (n?.lightBg && dark) return false;
    if (n?.calmBg && br.busy) return false;
    if (n?.gradientBg && !br.gradientLike) return false;
    if (br.busy && !s.allowedOnBusy) return false;
    if (s.hero && bgk.kind === 'animated') return false;   // one loud effect only
    return true;
  });
  return rng.weighted(ok, weightBy(ok, moods));
}

export function buildButtons(p: Palette, bgk: Theme['background'], s: SurfaceRecipe, moods: Mood[], fontCat: string, rng: RNG): Theme['buttons'] {
  const bg = bgk.color;
  const transparent = ['outline', 'glass', 'underline', 'neon-glow'].includes(s.surface);
  let fill: Hex = rng.chance(0.55) ? p.accent : p.text;
  if (s.surface === 'metallic') fill = p.accent;
  if (s.surface === 'soft-tint') fill = mix(bg, p.accent, rng.range(0.14, 0.24));
  if (s.surface === 'neumorphic') fill = bg;
  if (s.surface === 'sticker' || s.surface === '3d-offset') fill = rng.chance(0.5) ? p.accent2 : p.surface;
  if (s.surface === 'soft-shadow' || s.surface === 'solid') fill = rng.chance(0.5) ? p.text : p.accent;
  if (transparent) fill = 'transparent' as Hex;
  const textBase = transparent ? p.text : readableOn(fill, undefined, 7);
  let text = transparent ? readableOn(bg, p.text, 7) : textBase;
  if (s.surface === 'soft-tint' || s.surface === 'neumorphic') text = readableOn(fill, p.text, 7);

  const edge = ({ outline: p.accent, 'neon-glow': p.accent, glass: mix(p.text, bg, 0.6), underline: p.accent,
    sticker: p.text, '3d-offset': p.text } as Record<string, Hex>)[s.surface] ?? fill;
  const bw = ({ outline: rng.pick([1, 2]), 'neon-glow': 1, glass: 1, underline: 0, sticker: 2, '3d-offset': 2 } as Record<string, number>)[s.surface] ?? 0;
  const shape = rng.weighted(['square', 'rounded', 'pill'] as const, [
    moods.some(m => ['editorial', 'retro', 'bold'].includes(m)) ? 2 : 0.6,
    2,
    moods.some(m => ['playful', 'creator', 'calm'].includes(m)) ? 2.2 : 0.8]);
  const upper = moods.some(m => ['bold', 'futuristic', 'retro'].includes(m)) && rng.chance(0.3);
  const serif = fontCat === 'serif';
  return {
    shape, surface: s.surface, fill, fillTransparent: transparent, text, border: edge as Hex,
    borderWidth: bw as 0 | 1 | 2 | 3 | 4,
    height: moods.includes('business') || moods.includes('minimal') ? rng.pick([46, 56] as const) : rng.pick([56, 56, 68] as const),
    shadow: s.surface === 'soft-shadow' ? 'soft' : s.surface === '3d-offset' ? 'none' : moods.includes('bold') && rng.chance(0.25) ? 'deep' : 'none',
    weight: serif ? rng.pick([400, 500] as const) : upper ? rng.pick([600, 700] as const) : rng.pick([500, 600] as const),
    case: upper ? 'upper' : 'normal',
    tracking: upper ? 'wide' : moods.includes('luxe') ? rng.pick(['normal', 'wide'] as const) : 'normal',
    motion: rng.weighted(MOTIONS.map(m => m.v), weightBy(MOTIONS.map(m => ({ moods: m.moods })), moods)),
  };
}

const pickOpt = <T>(opts: Opt<T>[], moods: Mood[], u: UseCase | undefined, rng: RNG): T =>
  rng.weighted(opts.map(o => o.v), weightBy(opts, moods, u));

export function buildLayout(p: Palette, bgk: Theme['background'], moods: Mood[], u: UseCase | undefined, hero: boolean, rng: RNG): Theme['layout'] {
  const header = pickOpt(HEADERS, moods, u, rng);
  const m = (...x: Mood[]) => moods.some(y => x.includes(y));
  const gradientish = !!bgk.gradient;
  return {
    header,
    avatarFrame: pickOpt(AVATAR_FRAMES, moods, u, rng),
    avatarSize: header === 'hero' ? rng.pick([128, 160] as const) : header === 'compact-row' ? rng.pick([72, 96] as const) : rng.pick(AVATAR_SIZES.slice(1, 4)),
    avatarGlow: hero ? 'none' : m('futuristic', 'playful') && rng.chance(0.35) ? 'accent' : m('calm', 'luxe', 'warm') && rng.chance(0.3) ? 'soft' : 'none',
    linkLayout: rng.weighted(['stack', '2-column', 'chips'] as const, [7, m('business', 'creator') ? 1.5 : 0.5, m('minimal') ? 1.2 : 0.4]),
    spacing: m('luxe', 'calm', 'editorial') ? rng.weighted(['balanced', 'airy'] as const, [1, 2]) : m('business', 'minimal') ? rng.weighted(['compact', 'balanced', 'airy'] as const, [1, 2, 1]) : 'balanced',
    socialIcons: m('futuristic', 'luxe') && gradientish ? 'glass' : m('bold', 'playful') ? 'filled' : 'outline',
    cardLook: pickOpt(CARD_LOOKS, moods, u, rng),
    width: rng.weighted(['focused', 'standard', 'wide'] as const, [7, 2.5, 0.5]),
    entrance: rng.weighted(['none', 'fade-up', 'stagger-scale'] as const, [1.5, 6, m('playful', 'bold') ? 2.5 : 0.8]),
  };
}

// rules.ts: HARD rules (never break) + SOFT scoring (0..100)
import { contrast, badge, isDark, luminance } from './util';
import type { Theme } from './types';
import { FONTS } from './data-fonts';
import { SURFACES, BG_RECIPES } from './data-styles';

const fontCat = (n: string) => FONTS.find(f => f.name === n);
export const bgColors = (t: Theme): string[] =>
  t.background.gradient ? t.background.gradient.stops.map(s => s.color) : [t.background.color];
/** worst-case contrast of a colour against every background stop */
export const worstContrast = (c: string, t: Theme) => Math.min(...bgColors(t).map(b => contrast(b, c)));

export function hardViolations(t: Theme, plan: 'free' | 'pro' = 'free'): string[] {
  const v: string[] = [];
  const bg = t.background.kind;
  const sr = SURFACES.find(s => s.surface === t.buttons.surface)!;
  const br = BG_RECIPES.find(b => b.kind === bg)!;

  if (worstContrast(t.page.text, t) < 4.5) v.push('page text contrast < 4.5');
  if (worstContrast(t.page.muted, t) < 3) v.push('muted text contrast < 3');

  const ownFill = !t.buttons.fillTransparent;
  const btnBg = ownFill ? t.buttons.fill : null;
  if (btnBg) {
    if (contrast(btnBg, t.buttons.text) < 4.5) v.push('button text contrast < 4.5');
    if (worstContrast(btnBg, t) < 1.5 && !sr.hasOwnEdge && t.buttons.borderWidth === 0 && t.buttons.shadow === 'none')
      v.push('button fill blends with background');
    if (['gradient', 'metallic'].includes(t.buttons.surface) && contrast(t.page.accent2, t.buttons.text) < 4.5)
      v.push('gradient button end-colour contrast < 4.5');
  } else if (worstContrast(t.buttons.text, t) < 4.5) v.push('transparent button text contrast < 4.5');
  if (['outline', 'neon-glow', 'underline'].includes(t.buttons.surface) && worstContrast(t.buttons.border, t) < 3)
    v.push('button border not visible (< 3:1)');

  const fams = new Set([t.typography.name, t.typography.bio, t.typography.buttons]);
  if (fams.size > 3) v.push('>3 font families');
  for (const f of [t.typography.bio, t.typography.buttons]) {
    const ft = fontCat(f);
    if (!ft) v.push('unknown font ' + f);
    else if (ft.displayOnly || ft.cat === 'script' || ft.cat === 'display') v.push(`display font "${f}" used for body/buttons`);
  }
  if (!fontCat(t.typography.name)) v.push('unknown name font');

  const dark = isDark(t.background.color);
  const n = sr.needs;
  if (n?.darkBg && !dark) v.push(`${sr.surface} needs dark bg`);
  if (n?.lightBg && dark) v.push(`${sr.surface} needs light bg`);
  if (n?.calmBg && br.busy) v.push(`${sr.surface} needs calm bg`);
  if (n?.gradientBg && !br.gradientLike) v.push(`${sr.surface} needs gradient-like bg`);
  if (br.busy && !sr.allowedOnBusy) v.push('busy background needs solid/glass buttons');

  if (plan === 'free') {
    if (br.pro) v.push('pro background for free user');
    if (sr.pro) v.push('pro button surface for free user');
  }
  return v;
}

const overlap = (a: string[], b: string[]) => a.filter(x => b.includes(x)).length;

/** Soft score, 0..100. */
export function score(t: Theme, trioScore: number, useCase?: string, harmony = 80): number {
  const m = t.vibe.moods;
  const br = BG_RECIPES.find(b => b.kind === t.background.kind)!;
  const sr = SURFACES.find(s => s.surface === t.buttons.surface)!;
  const nameF = FONTS.find(f => f.name === t.typography.name)!;
  const moodHit = (xs: string[]) => Math.min(1, overlap(m, xs) / Math.min(2, m.length));
  const moodScore = (moodHit(br.moods) + moodHit(sr.moods) + moodHit(nameF.moods)) / 3 * 100;
  let s = 0.35 * moodScore + 0.25 * trioScore + 0.2 * harmony + 4; // +4 base consistency
  // consistency bonuses
  if (m.includes('editorial') && nameF.cat === 'serif') s += 3;
  if (m.includes('editorial') && ['underline', 'outline'].includes(t.buttons.surface)) s += 2;
  if (m.includes('futuristic') && ['mono', 'sans', 'display'].includes(nameF.cat)) s += 2;
  if (useCase) s += 2;
  // loud-effect penalty: max one hero effect
  const heroes = (sr.hero ? 1 : 0) + (t.background.kind === 'animated' ? 1 : 0) + (t.layout.avatarGlow === 'accent' ? 0.5 : 0);
  if (heroes > 1) s -= 12 * (heroes - 1);
  // readability bonus
  const cr = contrast(t.page.text, t.background.color);
  if (cr >= 7) s += 3;
  return Math.max(0, Math.min(100, Math.round(s)));
}

// generator.ts: generateTheme() + repair + section suggestions + regenerate
import { RNG, mix, shift, contrast, badge, rgbToHsl, readableOn, type Hex } from './util';
import { PALETTES, type Palette, type Mood, type UseCase } from './data-palettes';
import { VIBES, type Vibe } from './data-vibes';

export const MAX_TRIES = 24;
const clone = <T,>(x: T): T => JSON.parse(JSON.stringify(x));
export type Section = 'vibe' | 'background' | 'typography' | 'buttons' | 'layout';
const ov = (a: readonly string[], b: readonly string[]) => a.filter(x => b.includes(x)).length;
const hueOf = (c: Hex) => rgbToHsl(c)[0];
const hueDist = (a: Hex, b: Hex) => { const d = Math.abs(hueOf(a) - hueOf(b)) % 360; return d > 180 ? 360 - d : d; };

// ---------- smart context ----------
export function inferFromContext(ctx: Context): { moods: Mood[]; useCase?: UseCase } {
  const moods: Mood[] = ctx.mood ? [ctx.mood] : [];
  let useCase = ctx.useCase;
  const lt = (ctx.linkTypes ?? []).map(x => x.toLowerCase());
  const has = (...k: string[]) => lt.some(x => k.some(y => x.includes(y)));
  if (!useCase) {
    if (has('spotify', 'soundcloud', 'youtube', 'apple-music')) useCase = 'musician';
    else if (has('github', 'gitlab', 'codepen')) useCase = 'developer';
    else if (has('linkedin', 'calendly')) useCase = 'business';
    else if (has('instagram', 'tiktok', 'pinterest')) useCase = 'influencer';
    else if (has('behance', 'dribbble')) useCase = 'artist';
  }
  if (has('shop', 'store', 'gumroad', 'etsy', 'payment', 'coffee')) moods.push('bold');
  const bio = (ctx.bio ?? '').toLowerCase();
  if (/student|learn|study/.test(bio)) useCase ??= 'student';
  if (/design|artist|illustrat|photograph/.test(bio)) useCase ??= 'artist';
  if (/music|singer|producer|dj|rapper/.test(bio)) useCase ??= 'musician';
  if (/developer|engineer|code|software/.test(bio)) useCase ??= 'developer';
  return { moods, useCase };
}

function harmonyScore(p: Palette): number {
  const d = hueDist(p.accent, p.accent2);
  return d < 20 ? 86 : d < 60 ? 92 : d > 150 ? 90 : d > 100 ? 86 : 78;
}

function anchorPalette(ctx: Context): { palette: Palette; hueDelta: number } | null {
  const cur = ctx.current;
  if (!ctx.locked?.vibe || !cur?.vibe) return null;
  const v = VIBES.find(x => x.id === cur.vibe!.id);
  const pals = (v ? v.palettes.map(id => PALETTES.find(p => p.id === id)!) : PALETTES).filter(Boolean);
  const acc = cur.page?.accent;
  const palette = acc ? [...pals].sort((a, b) => hueDist(a.accent, acc) - hueDist(b.accent, acc))[0] : pals[0];
  const hueDelta = acc ? ((hueOf(acc) - hueOf(palette.accent) + 540) % 360) - 180 : 0;
  return { palette, hueDelta };
}

function pickVibe(rng: RNG, ctx: Context, plan: 'free' | 'pro'): Vibe {
  const { moods, useCase } = inferFromContext(ctx);
  const pool = VIBES.filter(v => !v.pro || plan === 'pro');
  const photo = ctx.photoColors?.[0];
  const w = pool.map(v => {
    let x = 1 + ov(v.moods, moods) * 2.5;
    if (useCase && v.useCases.includes(useCase)) x *= 2;
    if (photo) { const p = PALETTES.find(q => q.id === v.palettes[0])!; x *= 1 + Math.max(0, 1 - hueDist(p.accent, photo) / 90); }
    return x;
  });
  return rng.weighted(pool, w);
}

// ---------- repair: makes any theme pass the hard rules where possible ----------
export function repair(t: Theme): Theme {
  const bgs = t.background.gradient ? t.background.gradient.stops.map(s => s.color) : [t.background.color];
  const best = (cands: Hex[], pick: (c: Hex) => number) => cands.reduce((a, b) => (pick(b) > pick(a) ? b : a));
  const w = (c: Hex) => Math.min(...bgs.map(b => contrast(b, c)));
  if (w(t.page.text) < 4.5) t.page.text = best([t.page.text, '#FFFFFF', '#0B0A0A'], w);
  if (w(t.page.muted) < 3) { t.page.muted = mix(t.page.text, t.background.color, 0.3); if (w(t.page.muted) < 3) t.page.muted = t.page.text; }
  const b = t.buttons, sr = SURFACES.find(s => s.surface === b.surface)!;
  if (b.fillTransparent) {
    if (w(b.text) < 4.5) b.text = best([b.text, t.page.text, '#FFFFFF', '#0B0A0A'], w);
    if (['outline', 'neon-glow', 'underline'].includes(b.surface) && w(b.border) < 3) b.border = best([b.border, t.page.text, t.page.accent], w);
  } else {
    if (contrast(b.fill, b.text) < 4.5) b.text = readableOn(b.fill, t.page.text, 4.5);
    if (['gradient', 'metallic'].includes(b.surface) && contrast(t.page.accent2, b.text) < 4.5)
      t.page.accent2 = mix(t.page.accent2, b.fill, 0.65);
    if (w(b.fill) < 1.5 && !sr.hasOwnEdge && b.borderWidth === 0 && b.shadow === 'none') { b.borderWidth = 1; b.border = t.page.text; }
  }
  return t;
}

const LABEL: Record<string, string> = { 'soft-shadow': 'Soft shadow', 'soft-tint': 'Soft tint', '3d-offset': '3D offset', 'neon-glow': 'Neon glow' };
const cap = (s: string) => (LABEL[s] ?? s.replace(/-/g, ' ')).replace(/^./, c => c.toUpperCase());

function attempt(seed: string, ctx: Context, plan: 'free' | 'pro'): Theme {
  const rng = new RNG(seed);
  const { useCase } = inferFromContext(ctx);
  const anchor = anchorPalette(ctx);
  const vibe = ctx.locked?.vibe && ctx.current?.vibe ? (VIBES.find(v => v.id === ctx.current!.vibe!.id) ?? pickVibe(rng, ctx, plan)) : pickVibe(rng, ctx, plan);
  const palId = rng.pick(vibe.palettes);
  let pal = anchor?.palette ?? PALETTES.find(p => p.id === palId)!;
  if (plan === 'free' && pal.pro) pal = PALETTES.find(p => !p.pro)!;
  const dHue = anchor ? anchor.hueDelta : rng.chance(0.65) ? rng.range(-14, 14) : 0;
  const dL = anchor ? 0 : rng.range(-2.5, 2.5);
  const P: Palette = { ...pal, bg: shift(pal.bg, dHue, 0, dL), bg2: shift(pal.bg2, dHue, 0, dL), surface: shift(pal.surface, dHue, 0, dL),
    accent: shift(pal.accent, dHue), accent2: shift(pal.accent2, dHue) };
  const moods = vibe.moods;

  const background = buildBackground(P, moods, rng, plan);
  const trio = rng.weighted(TRIOS, TRIOS.map(t => (0.2 + ov(t.moods, moods)) * (t.score / 90) ** 4));
  const surface = pickSurface(P, background, moods, rng, plan);
  const fontCat = FONTS.find(f => f.name === trio.buttons)?.cat ?? 'sans';
  const buttons = buildButtons(P, background, surface, moods, fontCat, rng);
  const layout = buildLayout(P, background, moods, useCase, !!surface.hero, rng);

  const t: Theme = {
    vibe: { id: vibe.id, name: vibe.name, tagline: vibe.tagline, moods: vibe.moods },
    page: { mode: pal.mode, text: P.text, muted: P.muted, accent: P.accent, accent2: P.accent2, surface: P.surface },
    background, typography: { name: trio.name, bio: trio.bio, buttons: trio.buttons }, buttons, layout,
    meta: { seed, name: vibe.name, palette: pal.name, moods: vibe.moods, score: 0, why: '', contrast: { body: 0, button: 0, badge: 'Fail' }, dots: [] },
  };
  // locks: keep the user's locked sections exactly as they are
  const L = ctx.locked, C = ctx.current;
  if (C) {
    if (L?.vibe && C.vibe && C.page) { t.vibe = C.vibe; t.page = { ...C.page }; }
    if (L?.background && C.background) t.background = C.background;
    if (L?.typography && C.typography) t.typography = C.typography;
    if (L?.buttons && C.buttons) t.buttons = { ...C.buttons };
    if (L?.layout && C.layout) t.layout = C.layout;
  }
  repair(t);
  return finalize(t, trio.score, useCase, harmonyScore(pal));
}

function finalize(t: Theme, trioScore: number, useCase: string | undefined, harmony: number): Theme {
  const body = worstContrast(t.page.text, t);
  const btn = t.buttons.fillTransparent ? worstContrast(t.buttons.text, t) : contrast(t.buttons.fill, t.buttons.text);
  t.meta.contrast = { body: +body.toFixed(1), button: +btn.toFixed(1), badge: badge(Math.min(body, btn)) };
  t.meta.score = score(t, trioScore, useCase, harmony);
  t.meta.dots = [t.background.color, t.page.accent, t.page.accent2, t.page.text];
  t.meta.why = `${t.meta.palette} colours with ${t.typography.name} / ${t.typography.bio} and ${cap(t.buttons.surface).toLowerCase()} ${t.buttons.shape} buttons. ` +
    `Text contrast ${t.meta.contrast.body}:1 (${t.meta.contrast.badge}).`;
  return t;
}

// ---------- public API ----------
const cache = new Map<string, Theme>();
export function generateTheme(ctx: Context = {}): Theme {
  const plan = ctx.plan ?? 'free';
  const base = ctx.seed ?? Math.random().toString(36).slice(2, 10);
  const cacheable = !ctx.current && !ctx.locked && !ctx.photoColors && !ctx.linkTypes && !ctx.bio;
  const key = `${base}|${plan}|${ctx.mood ?? ''}|${ctx.useCase ?? ''}`;
  if (cacheable && cache.has(key)) return clone(cache.get(key)!);
  let bestT: Theme | null = null;
  for (let i = 0; i < MAX_TRIES; i++) {
    const t = attempt(`${base}:${i}`, ctx, plan);
    if (hardViolations(t, plan).length) continue;
    if (t.meta.score >= 80) { bestT = t; break; }
    if (!bestT || t.meta.score > bestT.meta.score) bestT = t;
  }
  const out = bestT ?? repair(attempt(`${base}:fallback`, { plan }, plan));
  out.meta.seed = base;
  if (cacheable) { cache.set(key, out); if (cache.size > 500) cache.delete(cache.keys().next().value!); }
  return clone(out);
}

export const signature = (t: Theme) =>
  [t.vibe.id, t.background.kind, t.background.color, t.typography.name, t.typography.buttons, t.buttons.surface, t.buttons.shape, t.layout.header, t.layout.avatarFrame].join('|');

export interface Suggestion { id: string; seed: string; section: Section | 'all'; name: string; tagline: string; badge: string; ratio: number; dots: Hex[]; score: number; why: string; patch: Partial<Theme>; }

function toSuggestion(section: Section | 'all', t: Theme): Suggestion {
  const ratio = Math.min(t.meta.contrast.body, t.meta.contrast.button);
  const common = { id: `${section}:${t.meta.seed}`, seed: t.meta.seed, section, badge: t.meta.contrast.badge, ratio, dots: t.meta.dots, score: t.meta.score, why: t.meta.why };
  switch (section) {
    case 'background': return { ...common, name: `${t.meta.palette} · ${t.background.name}`, tagline: `${t.background.kind.replace(/-/g, ' ')} · ${t.background.gradient ? t.background.gradient.type + ' ' + t.background.gradient.angle + '°' : t.background.color}${t.background.grain ? ' · grain ' + t.background.grain : ''}`, patch: { background: t.background, page: t.page, buttons: t.buttons } };
    case 'typography': return { ...common, name: `${t.typography.name} · ${t.typography.bio}`, tagline: t.vibe.moods.slice(0, 2).join(' · '), patch: { typography: t.typography, buttons: t.buttons } };
    case 'buttons': return { ...common, name: `${cap(t.buttons.surface)} · ${cap(t.buttons.shape)}`, tagline: `${t.buttons.fillTransparent ? 'line ' + t.buttons.border : t.buttons.fill} · ${t.buttons.weight}${t.buttons.case === 'upper' ? ' · UPPER' : ''} · ${t.buttons.height}px`, patch: { buttons: t.buttons, page: t.page } };
    case 'layout': return { ...common, name: `${cap(t.layout.header)} · ${cap(t.layout.spacing)}`, tagline: `${cap(t.layout.avatarFrame)} avatar · ${cap(t.layout.cardLook)} cards · ${t.layout.linkLayout}`, patch: { layout: t.layout } };
    default: return { ...common, name: t.vibe.name, tagline: t.vibe.tagline, patch: t };
  }
}

/** 5-8 ideas for ONE section, matched to the user's current choices in the other sections. */
export function suggestSection(section: Section, ctx: Context = {}, count = 6, recent: string[] = []): { items: Suggestion[]; signatures: string[] } {
  const all: Section[] = ['vibe', 'background', 'typography', 'buttons', 'layout'];
  const locked = { ...(ctx.locked ?? {}) } as NonNullable<Context['locked']>;
  if (ctx.current) for (const s of all) if (s !== section && s !== 'vibe') locked[s] = true;
  if (ctx.current && section !== 'vibe') locked.vibe = true;
  if (section !== 'vibe' && ctx.current) locked[section] = false;
  if (section === 'vibe' && ctx.locked?.vibe === undefined) locked.vibe = false;
  const items: Suggestion[] = [], sigs: string[] = [], seen = new Set(recent), seenVisible = new Set<string>();
  const base = ctx.seed ?? Math.random().toString(36).slice(2, 8);
  for (let n = 0; items.length < count && n < count * 16; n++) {
    const t = generateTheme({ ...ctx, locked, seed: `${base}:${section}:${n}` });
    const s = toSuggestion(section, t);
    const vis = s.name + '|' + s.tagline;           // what the user actually sees on the card
    const sg = vis + '|' + s.dots.join('');
    if (seen.has(sg) || seenVisible.has(vis)) continue;
    seen.add(sg); seenVisible.add(vis); sigs.push(sg); items.push(s);
  }
  return { items, signatures: sigs };
}

/** Refresh icon: new seed, never repeats the last 20 shown (client sends them back). */
export function regenerate(section: Section | 'all', ctx: Context = {}, recent: string[] = []) {
  const seed = Math.random().toString(36).slice(2, 10);
  const r = section === 'all'
    ? { items: Array.from({ length: 6 }, (_, i) => toSuggestion('all', generateTheme({ ...ctx, seed: `${seed}:${i}` }))), signatures: [] as string[] }
    : suggestSection(section, { ...ctx, seed }, 6, recent.slice(-20));
  if (section === 'all') r.signatures = r.items.map(s => s.name + '|' + s.tagline + '|' + s.dots.join(''));
  return r;
}

const bgCss = (t: Theme) => {
  const g = t.background.gradient;

  return g.type === 'linear' ? `linear-gradient(${g.angle}deg,${stops})` : g.type === 'radial' ? `radial-gradient(circle at 30% 20%,${stops})` : `conic-gradient(from ${g.angle}deg,${stops})`;
};
